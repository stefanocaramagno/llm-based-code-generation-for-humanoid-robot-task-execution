from reachy_sdk.trajectory import InterpolationMode

    # Coordinate della pallina da basket (leggermente a destra e sopra il tavolo)
    ball_position = np.array([0.10, 0.3187999, -0.01479991])  # Aggiustare x per la posizione a destra

    # 1. Far guardare il robot verso la pallina
    look_at(ball_position)

    # 2. Posizionare la pinza sopra la pallina
    above_ball = np.array([ball_position[0], ball_position[1], ball_position[2] + 0.1])  # Posizionare sopra la palla
    move_arm(above_ball, 2.0)

    # 3. Aprire la pinza
    open_gripper()

    # 4. Scendere il braccio fino a farlo poggiare sul tavolo
    on_table = np.array([ball_position[0], ball_position[1], ball_position[2] - 0.1])  # Poggiato sul tavolo
    move_arm(on_table, 2.0)

    # 5. Afferrare la palla
    close_gripper()

    # 6. Sollevare il braccio con la pallina
    move_arm(above_ball, 2.0)

    # 7. Riscendere il braccio fino a farlo poggiare sul tavolo
    move_arm(on_table, 2.0)

    # 8. Aprire la pinza per rilasciare la pallina
    open_gripper()

    # 9. Riportare il braccio alla posizione iniziale
    initial_position = np.array([0.4, 0.3187999, -0.01479991])  # Posizione iniziale del braccio
    move_arm(initial_position, 2.0)
